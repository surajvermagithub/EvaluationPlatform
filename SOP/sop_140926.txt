        ↓

CURRENT ApplicationDbContext BASELINE:

using EvaluationPlatform.Domain.Entities;
using Microsoft.EntityFrameworkCore;

namespace EvaluationPlatform.Infrastructure.Data;

public class ApplicationDbContext : DbContext
{
    public ApplicationDbContext(DbContextOptions<ApplicationDbContext> options)
        : base(options)
    {
    }

    public DbSet<User> Users => Set<User>();
    public DbSet<Role> Roles => Set<Role>();
    public DbSet<Institute> Institutes => Set<Institute>();

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        modelBuilder.Entity<User>()
            .HasOne(u => u.Institute)
            .WithMany(i => i.Users)
            .HasForeignKey(u => u.InstituteId)
            .OnDelete(DeleteBehavior.Restrict);

        modelBuilder.Entity<User>()
            .HasOne(u => u.Role)
            .WithMany(r => r.Users)
            .HasForeignKey(u => u.RoleId)
            .OnDelete(DeleteBehavior.Restrict);

        // Fluent API validation/constraints also exist.
        // Institute.Code should be unique.
    }
}

CURRENT USER REPOSITORY

Infrastructure/Repositories/UserRepository.cs

Current important methods:

public async Task<bool> SuperAdminExistsAsync()
{
    return await _context.Users
        .AnyAsync(u => u.Role.Name == "SuperAdmin" && !u.IsDeleted);
}

public async Task<User?> GetByEmailAsync(string email)
{
    return await _context.Users
        .FirstOrDefaultAsync(u => u.Email == email && !u.IsDeleted);
}

Role lookup was added conceptually:

public async Task<Role?> GetRoleByNameAsync(string roleName)
{
    return await _context.Roles
        .FirstOrDefaultAsync(r =>
            r.Name == roleName &&
            !r.IsDeleted);
}

Add:

public async Task<User> AddAsync(User user)
{
    await _context.Users.AddAsync(user);
    await _context.SaveChangesAsync();
    return user;
}

==================================================
IMPORTANT ERROR WE ALREADY SOLVED
==================================================

During first registration, we got a DbUpdateException caused by a foreign-key problem.

Reason:
User was being inserted with:

RoleId = 1

but Role ID 1 did not exist in Roles table.

We then seeded roles using HasData.

Also noticed that first User ID was 5 instead of 1.

Explanation:
SQL Server Identity columns do NOT guarantee gapless IDs.
Previous inserts/deletes/migrations can cause the next ID to be 5.

This is normal.

Do NOT reset identity just because IDs are not consecutive.
